cmake_minimum_required(VERSION 3.16)
project(zeref_r12_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(zeref_r12_lib src/sha256.cpp src/r12.cpp)
target_include_directories(zeref_r12_lib PUBLIC include)
add_executable(zeref-r12-native src/main.cpp)
target_link_libraries(zeref-r12-native PRIVATE zeref_r12_lib)
enable_testing()
add_executable(zeref-r12-tests tests/test_r12.cpp)
target_link_libraries(zeref-r12-tests PRIVATE zeref_r12_lib)
add_test(NAME zeref-r12-native-tests COMMAND zeref-r12-tests)
